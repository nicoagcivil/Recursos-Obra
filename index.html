function getIconDataForEquipo(sub) { const n = sub.toLowerCase(); if (n.includes("excavadora") || n.includes("cargadora")) { return { type: "img", src: "excavadora.jpg", fallbackIcon: "fa-bucket" }; } if (n.includes("rodillo")) { return { type: "img", src: "rodillo.jpg", fallbackIcon: "fa-compact-disc" }; } if (n.includes("motoniveladora") || n.includes("topadora")) { return { type: "img", src: "motoniveladora.jpg", fallbackIcon: "fa-road" }; } if (n.includes("luminaria")) { return { type: "img", src: "luminaria.jpg", fallbackIcon: "fa-lightbulb" }; } if (n.includes("fuera de ruta")) { return { type: "img", src: "camion fuera de ruta.jpg", fallbackIcon: "fa-truck-monster" }; } if (n.includes("camioneta")) { return { type: "img", src: "camioneta.jpg", fallbackIcon: "fa-truck-pickup" }; } if (n.includes("camión") || n.includes("camion") || n.includes("regador")) { return { type: "img", src: "camion.jpg", fallbackIcon: "fa-truck-field" }; } // Iconos de respaldo para el resto if (n.includes("combustible") || n.includes("lubricante")) return { type: "fa", icon: "fa-gas-pump" }; if (n.includes("manipulador")) return { type: "fa", icon: "fa-boxes-packing" }; if (n.includes("combi")) return { type: "fa", icon: "fa-bus" }; if (n.includes("generador")) return { type: "fa", icon: "fa-bolt" }; return { type: "img", src: "camion.jpg", fallbackIcon: "fa-truck-monster" }; }
